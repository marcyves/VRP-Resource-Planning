\section{Aide-mémoire}
\label{sec:memoire}

\subsection{Checklist premier jour}

\begin{itemize}[label=$\square$, leftmargin=*]
  \item Connexion OK, profil \emph{Médical} confirmé
  \item Au moins une \prestation{} et une \structure{}
  \item Une \seance{} et un \patient{} de test
  \item Une \seance{} planifiée dans l'agenda
  \item Parcours facturation \texttt{\#billing} parcouru sur la \structure{}
  \item Compte bancaire renseigné (Trésorerie → Banque)
\end{itemize}

\subsection{Checklist hebdomadaire}

\begin{itemize}[label=$\square$, leftmargin=*]
  \item \seance{}s de la semaine saisies ou importées
  \item Factures du mois préparées par \structure{}
  \item Relevé bancaire importé et rapproché
  \item Montants \emph{non facturé TTC} sur l'accueil à zéro ou expliqués
\end{itemize}

\subsection{Dépannage rapide}

\begin{table}[htbp]
  \centering
  \begin{tabular}{@{}lp{7cm}@{}}
    \toprule
    \textbf{Problème} & \textbf{Piste} \\
    \midrule

\end{table}
