% Capture d'écran
\newcommand{\shotpath}{screenshots/}
\newcommand{\screenshot}[3][]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{\shotpath#2}{%
      \includegraphics[width=\linewidth,#1]{\shotpath#2}%
    }{%
      \begin{tikzpicture}
        \node[
          draw=vrpmuted,
          dashed,
          rounded corners=4pt,
          minimum width=0.92\linewidth,
          minimum height=5.5cm,
          align=center,
          text=vrpmuted,
          font=\small
        ] {Capture non générée\\\texttt{\shotpath#2}\\[4pt]
           \footnotesize Lancer \texttt{node scripts/capture-screenshots.mjs}};
      \end{tikzpicture}%
    }
    \caption{#3}
    \label{fig:#2}
  \end{figure}%
}
